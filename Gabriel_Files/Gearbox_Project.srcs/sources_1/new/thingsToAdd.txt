# HC-020K speed sensor input (Pmod JD1)
set_property PACKAGE_PIN R2 [get_ports sensor_in]
set_property IOSTANDARD LVCMOS33 [get_ports sensor_in]





#You can instantiate the sensor inside main_motor_top_pwm
-- Add ports to main_motor_top_pwm
entity main_motor_top_pwm is
    Port (
        clk      : in  STD_LOGIC;
        btnC     : in  STD_LOGIC;
        sw       : in  STD_LOGIC_VECTOR(15 downto 0);
        sensor_in: in  STD_LOGIC;                        -- new
        JA       : out STD_LOGIC_VECTOR(3 downto 2);
        JB       : out STD_LOGIC_VECTOR(3 downto 0);
        JC       : out STD_LOGIC_VECTOR(3 downto 0);
        -- optional: display speed on LEDs
        led      : out STD_LOGIC_VECTOR(15 downto 0)     -- new (shows rpm)
    );
end main_motor_top_pwm;

#then inside Architecture:
-- Instantiate speed sensor
signal rpm_value   : std_logic_vector(15 downto 0);
signal speed_valid : std_logic;

U_speed_sensor : entity work.hc020k_speed_sensor
    generic map (
        CLK_FREQ_HZ    => 100_000_000,
        MEASUREMENT_MS => 100,
        SLOTS_PER_REV  => 20      -- adjust to your encoder disc
    )
    port map (
        clk       => clk,
        rst       => btnC,
        sensor_in => sensor_in,
        pulse_hz  => open,
        rpm       => rpm_value,
        valid     => speed_valid
    );

-- Display RPM on LEDs (optional)
process(clk)
begin
    if rising_edge(clk) then
        if speed_valid = '1' then
            led <= rpm_value;
        end if;
    end if;
end process;
